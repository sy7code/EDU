const StateBlock = ({ state, onRetry }: { state: DataState; onRetry?: () => void }) => {
  if (state === "loading")
    return (
      <div className="state-block">
        <span className="spinner" />
        <strong>불러오는 중</strong>
        <p>기록을 안전하게 가져오고 있어요.</p>
      </div>
    );
  if (state === "empty")
    return (
      <div className="state-block">
        <span className="state-icon">
          <Icon name="book" />
        </span>
        <strong>아직 기록이 없어요</strong>
        <p>첫 학습을 시작하면 여기에 차곡차곡 쌓여요.</p>
      </div>
    );
  if (state === "error")
    return (
      <div className="state-block">
        <span className="state-icon error">
          <Icon name="alert" />
        </span>
        <strong>불러오지 못했어요</strong>
        <p>잠시 후 다시 시도해 주세요.</p>
        <Button variant="secondary" onClick={onRetry}>
          다시 시도
        </Button>
      </div>
    );
  return null;
};

export default function App() {
  const [role, setRole] = useState<Role>("student");
  const [view, setView] = useState<View>("home");
  const [dataState, setDataState] = useState<DataState>("ready");
  const changeRole = (r: Role) => {
    setRole(r);
    setView(r === "student" ? "home" : "dashboard");
  };
  return (
    <div className={`app-shell ${view === "login" ? "is-login" : ""}`}>
      {view !== "login" && (
        <>
          <AppHeader role={role} setRole={changeRole} onLogout={() => setView("login")} />
          <Sidebar role={role} view={view} setView={setView} setRole={changeRole} />
        </>
      )}
      <main>
        {view !== "login" && (
          <div className="state-demo" title="목록 상태 미리보기">
            <span>상태</span>
            {(["ready", "loading", "empty", "error"] as DataState[]).map((x) => (
              <button
                className={dataState === x ? "active" : ""}
                onClick={() => setDataState(x)}
                key={x}
              >
                {x === "ready"
                  ? "기본"
                  : x === "loading"
                    ? "로딩"
                    : x === "empty"
                      ? "비어 있음"
                      : "오류"}
              </button>
            ))}
          </div>
        )}
        {view === "login" && <Login onLogin={() => setView("home")} />}
        {view === "home" && <StudentHome setView={setView} dataState={dataState} />}
        {view === "study" && <Study />}
        {view === "records" && <Records dataState={dataState} />}
        {view === "consent" && <Consent />}
        {view === "dashboard" && <Dashboard dataState={dataState} setView={setView} />}
        {view === "analysis" && <Analysis />}
        {view === "student" && <StudentDetail />}
        {view === "reports" && <Reports />}
      </main>
      <div className="toast" role="status">
        <Icon name="check" />
        <span>
          <strong>저장되었어요</strong>변경한 내용이 안전하게 저장되었습니다.
        </span>
        <button aria-label="알림 닫기">×</button>
      </div>
    </div>
  );
}
